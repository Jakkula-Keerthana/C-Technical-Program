#include<stdio.h>
int main()
{
int n;
printf("enter the n value");
 scanf("%d",&n);
for(int i=0;i<n;i++){
for(int j=0;j<n;j++){
printf("__");
}
for(int k=0;k<2*i+1;k++){
printf("*");
}
printf("\n");
}
for(int i=n;i>=0;i--){
for(int j=0;j<(n-i)-1;j++){
printf("_");
}
for(int k=0;k<2*i+1;k++){
printf("*");
}
printf("\n");
}
}
